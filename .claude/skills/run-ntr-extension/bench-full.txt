# Driver commands that turn on most of NTR for `st.sh bench <name> [ref] full`. st.sh replaces @BANNERS@ with three
# banner images and @IMG@ with the folder of test images. The banner and foreground images go through the menu;
# the other settings are set directly, then the page reloads to check they were saved.
select default_Seraphina.png
menu
click #cb_modal_overlay #m_b_enable
click #cb_modal_overlay #m_b_up
upload #cb_modal_overlay #m_b_file | @BANNERS@
wait 2000
click #cb_modal_overlay #m_b_rot
menu fg
click #cb_modal_overlay #m_f_enable
click #cb_modal_overlay .m_f_up[data-pos="Left"]
upload #cb_modal_overlay #m_f_file | @IMG@/fg.png
wait 1500
click #cb_modal_overlay .m_f_up[data-pos="Right"]
upload #cb_modal_overlay #m_f_file | @IMG@/fg.png
wait 1500
close
eval (() => { const A = NTR.api, cur = A.store().fg.Left; Object.assign(A.settings(), { bannerRotateSec: 3, ovAvatarOn: true, ovAvatar: 'rounded', rbBtnColorOn: true, rbBtnColor: '#7a3cff', rbBtnFx: 'glow', rbEdgeFx: 'glow', rbBoxColorOn: true, rbBoxColor: '#20103a', rbBoxAlphaOn: true, rbBoxOpacity: 70, rbBorderStyleOn: true, rbBorderStyle: 'solid', rbPat: 'notebook', rbPatV: true, rbTyping: 'both', tfNameFxOn: true, tfAiFxOn: true, tfUserFxOn: true, tfAiMainOn: true, tfAiMain: '#e8d8ff', tfAiQuoteOn: true, tfAiQuote: '#ffb3e6', tfNameFontOn: true, tfNameFont: 'Georgia', ovWidthOn: true, ovWidth: 60, ovBlurOn: true, ovBlur: 10, ovShadowOn: true, ovShadow: 3, ovShapeOn: true, ovShape: 'rounded', ovRound: 16, ovPanelBg: 'color', ovPanelBgColor: '#140a24', ovPanelBgOpacity: 60, ovPanelBorderOn: true, ovMesGapOn: true, ovMesGap: 12, ovSendPosOn: true, ovScrollColorOn: true, ovScrollColor: '#7a3cff', ovCursorOn: true, ovCursorImg: cur, ovCursorPtrImg: cur, ovCursorSize: 32 }); A.save(); return 'settings saved'; })()
wait 3000
reload
select default_Seraphina.png
