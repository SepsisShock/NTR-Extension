# Driver commands that turn on most of NTR for `st.sh bench <name> [ref] full`. st.sh replaces @BANNERS@ with three
# banner images and @IMG@ with the folder of test images. The banner and foreground images go through the menu;
# the other settings are set directly, then the page reloads to check each one was saved and kept. The Pinned Block matches nothing,
# so it looks through every message each time the chat changes: the most work it can do. NTR adds its own defaults to
# what's set (the Pinned Block's other fields), so the check after the reload only looks at the values set here.
select default_Seraphina.png
menu
click #cb_modal_overlay #m_b_enable
click #cb_modal_overlay #m_b_up
upload #cb_modal_overlay #m_b_file | @BANNERS@
wait 2000
click #cb_modal_overlay #m_b_rot
menu fg
click #cb_modal_overlay #m_f_enable
click #cb_modal_overlay .m_f_up[data-pos="Left"]
upload #cb_modal_overlay #m_f_file | @IMG@/fg.png
wait 1500
click #cb_modal_overlay .m_f_up[data-pos="Right"]
upload #cb_modal_overlay #m_f_file | @IMG@/fg.png
wait 1500
close
